1"""Provider exposing FastMCP request objects to dishka dependencies. 

2 

3Add ``FastMCPProvider()`` to the container to let dependencies declare the 

4current ``Context``, the ``FastMCP`` server, or the raw request params of the 

5operation in flight. Every entry is ``Scope.REQUEST`` and resolved lazily, so 

6only the object that actually exists for the current operation is required — 

7a resource handler asking for ``ReadResourceRequestParams`` works; asking for 

8``CallToolRequestParams`` during a resource read does not. 

9""" 

10 

11from dishka import Provider, Scope, from_context 

12from fastmcp import FastMCP 

13from fastmcp.server.context import Context 

14from mcp import types as mt 

15 

16__all__ = ('FastMCPProvider',) 

17 

18 

19class FastMCPProvider(Provider): 

20 """Exposes FastMCP request-scoped objects via ``from_context``.""" 

21 

22 context = from_context(Context, scope=Scope.REQUEST) 

23 server = from_context(FastMCP, scope=Scope.REQUEST) 

24 call_tool_params = from_context(mt.CallToolRequestParams, scope=Scope.REQUEST) 

25 read_resource_params = from_context(mt.ReadResourceRequestParams, scope=Scope.REQUEST) 

26 get_prompt_params = from_context(mt.GetPromptRequestParams, scope=Scope.REQUEST)
